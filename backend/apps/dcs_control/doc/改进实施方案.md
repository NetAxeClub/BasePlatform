```python
from django.conf import settings

def sec_callback(flow_record_id):
    """
    流程完成后的外部回调通知。
    回调地址和触发用户通过 settings.DCS_CALLBACK_CONFIG 配置。
    """
    callback_config = getattr(settings, 'DCS_CALLBACK_CONFIG', None)
    if not callback_config:
        return  # 未配置则跳过

    flow_record = AutoFlow.objects.filter(id=flow_record_id).values().first()
    if not flow_record:
        log.warning("sec_callback: flow_record %s 不存在", flow_record_id)
        return

    allowed_users = callback_config.get('allowed_users', [])
    callback_url = callback_config.get('url', '')

    if flow_record['commit_user'] not in allowed_users:
        return
    if not callback_url:
        log.warning("sec_callback: DCS_CALLBACK_CONFIG.url 未配置")
        return

    try:
        response = requests.post(callback_url, json=flow_record, timeout=10)
        response.raise_for_status()
        log.info("sec_callback 成功: flow_id=%s, status=%s", flow_record_id, response.status_code)
    except requests.RequestException as e:
        log.error("sec_callback 失败: flow_id=%s, error=%s", flow_record_id, str(e))
```

### TODO-06：提取 TTP 错误解析公共函数

**背景**：`SecFirewallMain.flow_engine`（约第 416-460 行）和 `FirewallMain.flow_engine`（约第 635-670 行）的 TTP 解析逻辑完全相同（约 40 行）。

**新建文件**：`apps/dcs_control/utils/ttp_parser.py`
```python
# -*- coding: utf-8 -*-
"""TTP 解析结果处理工具"""
import logging

log = logging.getLogger(__name__)


def extract_ttp_error(ttp_res: dict) -> str:
    """
    从 TTP 解析结果字典中提取错误描述字符串。
    返回空字符串表示无错误。
    """
    if not isinstance(ttp_res, dict):
        return ''

    if 'unrecognized' in ttp_res:
        val = ttp_res['unrecognized']
        if isinstance(val, dict):
            return 'unrecognized: ' + str(val.get('unrecognized', val))
        if isinstance(val, list):
            return 'unrecognized: ' + '\n'.join(
                str(x.get('unrecognized', x)) for x in val)
        return 'unrecognized: ' + str(val)

    if 'incomplete' in ttp_res:
        val = ttp_res['incomplete']
        if isinstance(val, dict):
            return 'incomplete: ' + str(val.get('incomplete', val))
        if isinstance(val, list):
            return 'incomplete: ' + '\n'.join(
                str(x.get('incomplete', x)) for x in val)
        return 'incomplete: ' + str(val)

    if 'errors' in ttp_res:
        val = ttp_res['errors']
        if isinstance(val, dict):
            return 'error: ' + str(val.get('error', val))
        if isinstance(val, list):
            unique_errors = list(set(str(x.get('error', x)) for x in val))
            return 'error: ' + '\n'.join(unique_errors)
        return 'error: ' + str(val)

    if 'warning' in ttp_res:
        val = ttp_res['warning']
        return 'warning: ' + (str(val.get('error', val)) if isinstance(val, dict) else str(val))

    if 'wrong_parameter' in ttp_res:
        val = ttp_res['wrong_parameter']
        return 'wrong_parameter: ' + (
            str(val.get('wrong_parameter', val)) if isinstance(val, dict) else str(val))

    return ''


def check_ttp_failed(ttp_res: dict) -> tuple:
    """
    检查 TTP 结果是否包含错误。
    返回 (is_failed: bool, error_msg: str)
    """
    error_keys = ('unrecognized', 'incomplete', 'errors', 'warning', 'wrong_parameter')
    for key in error_keys:
        if key in ttp_res:
            return True, extract_ttp_error(ttp_res)
    return False, ''
```

### TODO-07：重构设备锁为 Redis 原子锁

**背景**：
- `run_cmd_config`（SSH）和 `run_netconf_config`（Netconf）均使用 `cache.get + cache.set` 非原子锁
- `time.sleep(30)` 阻塞 Celery Worker 线程（每次最多浪费 90 秒）

**新建文件**：`apps/dcs_control/utils/device_lock.py`
```python
# -*- coding: utf-8 -*-
"""设备操作原子锁"""
import logging
from contextlib import contextmanager
from django.core.cache import cache

log = logging.getLogger(__name__)

LOCK_TIMEOUT = 300  # 锁最大持有时间（秒），防止崩溃导致死锁


class DeviceBusyError(Exception):
    """设备当前被其他任务占用"""


@contextmanager
def device_lock(ip: str, timeout: int = LOCK_TIMEOUT):
    """
    原子设备锁上下文管理器。

    cache.add 在 Redis 后端是原子操作：key 不存在时才写入并返回 True。
    相比 get + set，消除了两次操作之间的竞态窗口。

    使用方式：
        with device_lock('192.168.1.1'):
            # 只有获得锁的任务才会执行此处
            do_config(...)
    """
    lock_key = f'dcs_device_lock_{ip}'
    acquired = cache.add(lock_key, '1', timeout=timeout)
    if not acquired:
        log.warning("设备 %s 当前被占用，拒绝新操作", ip)
        raise DeviceBusyError(f"设备 {ip} 正在被操作中，请稍后重试（锁超时 {timeout}s）")
    log.debug("设备锁已获取: %s", ip)
    try:
        yield
    finally:
        cache.delete(lock_key)
        log.debug("设备锁已释放: %s", ip)
```

**修改 `tasks.py` 中两个执行函数**：

### TODO-09：引入 `Vendor` 枚举统一厂商标识

**新建文件**：`apps/dcs_control/constants.py`
```python
# -*- coding: utf-8 -*-
"""DCS Control 模块常量定义"""
from enum import Enum


class Vendor(str, Enum):
    """防火墙厂商标识，统一大小写格式"""
    HILLSTONE = 'Hillstone'
    H3C = 'H3C'
    HUAWEI = 'Huawei'

    @classmethod
    def normalize(cls, raw: str) -> str:
        """将任意大小写输入转为规范格式"""
        mapping = {v.lower(): v.value for v in cls}
        return mapping.get(raw.lower(), raw)

    @classmethod
    def choices(cls):
        return [v.value for v in cls]
```

### TODO-11：`bulk_deny_by_address` 改为 Celery `group` 并行

**背景**：当前实现是在 for 循环中逐一 `apply_async`，虽然 Celery 会并行消费，但整个 `bulk_deny` Task 本身依赖 for 循环完成，且无法统一等待所有子任务结果。

**改后核心逻辑**：
```python
from celery import group

@shared_task(base=AxeTask, once={'graceful': True}, bind=True)
def bulk_deny_by_address(self, **post_param):
    hosts = AutomationInventory.objects.prefetch_related(...).filter(...).values(...)
    event_id = AutoEvent.objects.create(...)

    # 构建所有子任务参数列表
    sub_tasks = []
    for host in hosts:
        if not host['ans_group_hosts__ans_vars']:
            continue
        try:
            host_vars = json.loads(host['ans_group_hosts__ans_vars'])
        except (json.JSONDecodeError, TypeError):
            host_vars = ast.literal_eval(host['ans_group_hosts__ans_vars'])
            if isinstance(host_vars, str):
                host_vars = json.loads(host_vars)

        # 根据操作类型构建 post_data（提取公共字段构建函数）
        post_data = _build_deny_post_data(
            host=host,
            host_vars=host_vars,
            post_param=post_param,
            event_id=event_id.id,
            task_id=str(self.request.id),
        )
        if post_data:
            sub_tasks.append(address_set.s(**post_data))

    if sub_tasks:
        # 并行分发，不阻塞等待结果
        job = group(sub_tasks)
        job.apply_async(queue=CELERY_QUEUE)
        log.info("bulk_deny_by_address: 已分发 %d 个子任务", len(sub_tasks))
    return


def _build_deny_post_data(host, host_vars, post_param, event_id, task_id):
    """提取公共构建逻辑，避免重复字典拼接"""
    base = {
        "origin": post_param['origin'],
        "vendor": host_vars['vendor__alias'],
        "name": host['ans_group_hosts__ans_obj'],
        "hostip": host_vars['manage_ip'],
        "hostid": host_vars['id'],
        "user": post_param['user'],
        "remote_ip": post_param['remote_ip'],
        "room_group_name": post_param.get('room_group_name'),
        "task_id": task_id,
        "task": AutoFlowTasks.DENY,
        "event_id": event_id,
    }
    if post_param.get('ip_mask'):
        if post_param.get('add_detail_ip'):
            return {**base, "add_detail_ip": True, "ip_mask": post_param['ip_mask']}
        if post_param.get('del_detail_ip'):
            return {**base, "del_detail_ip": True, "ip_mask": post_param['ip_mask']}
    if post_param.get('range_start') and post_param.get('range_end'):
        if post_param.get('add_detail_range'):
            return {**base, "add_detail_range": True,
                    "range_start": post_param['range_start'], "range_end": post_param['range_end']}
        if post_param.get('del_detail_range'):
            return {**base, "del_detail_range": True,
                    "range_start": post_param['range_start'], "range_end": post_param['range_end']}
    return None
```

**收益**：代码行数从约 120 行降至约 50 行；逻辑清晰；消除 4 处重复的 post_data 字典拼接。

### TODO-13：策略模式重构多厂商适配

**新建文件**：`apps/dcs_control/firewall/base.py`
```python
# -*- coding: utf-8 -*-
from abc import ABC, abstractmethod


class FirewallAdapter(ABC):
    """
    防火墙适配器抽象基类。
    新增厂商只需继承本类实现所有方法，并在 registry.py 中注册即可。
    """

    def __init__(self, dev_info: dict, dev_infos: dict):
        self.dev_info = dev_info
        self.dev_infos = dev_infos

    # ── 地址对象 ─────────────────────────────────────
    @abstractmethod
    def build_address_cmds(self, **kwargs) -> tuple:
        """生成地址对象操作命令。返回 (cmds, back_off_cmds)"""

    # ── 服务对象 ─────────────────────────────────────
    @abstractmethod
    def build_service_cmds(self, **kwargs) -> tuple:
        """生成服务对象操作命令。返回 (cmds, back_off_cmds)"""

    # ── DNAT ─────────────────────────────────────────
    @abstractmethod
    def build_dnat_cmds(self, **kwargs) -> tuple:
        """生成 DNAT 操作命令。返回 (cmds, back_off_cmds)"""

    # ── 安全策略 ─────────────────────────────────────
    @abstractmethod
    def build_sec_policy_cmds(self, **kwargs) -> tuple:
        """生成安全策略命令。返回 (cmds, back_off_cmds)"""

    # ── 协议标识 ─────────────────────────────────────
    @property
    @abstractmethod
    def protocol(self) -> str:
        """返回下发协议: 'SSH' 或 'NETCONF'"""

    @property
    @abstractmethod
    def class_method(self) -> str:
        """返回 Netconf 类方法名（SSH 协议返回 'Hillstone'）"""
```

**新建文件**：`apps/dcs_control/firewall/registry.py`
```python
# -*- coding: utf-8 -*-
from apps.dcs_control.firewall.base import FirewallAdapter

_REGISTRY: dict[str, type] = {}


def register(vendor: str):
    """适配器注册装饰器"""
    def decorator(cls):
        _REGISTRY[vendor] = cls
        return cls
    return decorator


def get_adapter(vendor: str, dev_info: dict, dev_infos: dict) -> FirewallAdapter:
    cls = _REGISTRY.get(vendor)
    if not cls:
        raise ValueError(f"不支持的厂商: {vendor}，已注册厂商: {list(_REGISTRY.keys())}")
    return cls(dev_info, dev_infos)
```

**使用示例**（`tasks/address.py` 中）：
```python
from apps.dcs_control.firewall.registry import get_adapter
from apps.dcs_control.engine.flow import FlowEngine

@shared_task(...)
def address_set(self, **post_param):
    fw = FirewallMain(post_param['hostip'])     # 仍负责登录信息获取
    adapter = get_adapter(post_param['vendor'], fw.dev_info, fw.dev_infos)
    cmds, back_off_cmds = adapter.build_address_cmds(**post_param)
    FlowEngine(fw).run(cmds, back_off_cmds, adapter, **_build_flow_data(self, post_param))
```
